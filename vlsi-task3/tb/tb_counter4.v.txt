module tb_counter4;
    reg clk, reset;
    wire [3:0] count;
    counter4 uut (.clk(clk), .reset(reset), .count(count));
    always #5 clk = ~clk;
    initial begin
        clk = 0; reset = 1;
        #10 reset = 0;
        #160;
        #10 reset = 1;
        #10 reset = 0;
        #50 $finish;
    end
endmodule
